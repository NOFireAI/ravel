# Subquery corpus (ticket P10, docs/promql-evaluator-plan.md section 8):
# `expr[range:step]` evaluation against
# crates/ravel-promql-difftest/src/generator.rs's fixed dataset. Every
# offset is relative to the dataset's base_ts_ms, so a corpus file never
# encodes an absolute timestamp, except where the query text itself
# requires one for the `@` modifier (matching selectors.txt's own
# precedent: the absolute value is base_ts_ms/1000 plus a stated offset).
#
# `diff_gauge_walk` and `diff_counter_total{shape="reset"}` are both
# regularly sampled every 30s across the dataset's full ~1170s span
# (generator.rs's `regular_timestamps`), so every subquery window below
# stays inside real sample coverage.
#
# `mode: unordered` is correct for every success entry here, for the same
# reason as selectors.txt (ADR-0021 decision 3).

name: instant_rate_over_subquery_of_a_recording_style_expression
query: rate(sum(diff_counter_total{shape="reset"})[5m:1m])
kind: instant
time: +900s
mode: unordered

time: +1080s

name: instant_subquery_step_defaults_to_the_evaluation_interval

name: instant_subquery_grid_start_is_epoch_aligned_not_window_relative
# base_ts_ms (1_700_000_000_000 ms) is 20s past the nearest absolute
# minute boundary, so this subquery's nominal (window-relative) start
# never lands on a 1m-epoch multiple: the point count only comes out
# right if both engines align the grid to the epoch, not to the window.
query: count_over_time(diff_gauge_walk[3m:1m])
kind: instant
time: +600s
mode: unordered

# ADR-0025 allowlist (arithmetic residue, same class as rate.txt's
# rate/deriv/predict_linear entries): avg_over_time sums the subquery
# grid's values then divides by the count, using only IEEE 754 +/- and /,
# so any gap is arithmetic residue, not a transcendental libm difference.
# Measured up to 5 ULPs across this range's grid points (issue #224
# mismatch #2); the representative pair there, -4.242539167465618 vs
# -4.242539167465619, is 1 ULP by f64::to_bits, and only that one grid
# point is observable from the issue text.
# tolerance: 8 is the next power of two above the 5-ULP worst case;
# tighten it if a live run shows a smaller observed max (ADR-0025
# Consequences: a tightenable tolerance is a signal worth checking).
name: range_subquery_over_a_stepped_grid
query: avg_over_time(diff_gauge_walk[3m:1m])
kind: range
start: +300s
end: +900s
step: 60s
mode: unordered
tolerance: 8

# ADR-0025 allowlist (arithmetic residue, same class as rate.txt's
# rate/deriv/predict_linear entries): avg_over_time sums then divides,
# IEEE 754 arithmetic only, so this is residue rather than a libm
# difference. Measured 1 ULP: the exact values in issue #224 mismatch #3
# (-3.4712031303800788 vs -3.4712031303800783) are adjacent doubles by
# f64::to_bits. This is a single, fully observed value (one instant
# result), not a sampled worst case, so the measurement is exact.
# tolerance: 2 is 1 ULP of headroom, matching rate.txt's 1-ULP entries.
name: instant_subquery_at_modifier_pins_the_grids_own_end
query: avg_over_time(diff_gauge_walk[3m:1m] @ 1700000600.000)
kind: instant
time: +0s
mode: unordered
tolerance: 2

# Cap-exceeded entries: Ravel enforces a resolution cap (11_000 points,
# docs/query-engine.md "Budgets", issue #77) independently at every
# subquery evaluation node, before that node's grid is built. Both
# entries below ask for a single node's grid alone to exceed it, one
# directly and one through one level of nesting.
#
# Verified against the pinned Prometheus v3.13.1 binary (issue #224
# mismatches #4 and #5): Prometheus SUCCEEDS on both, Ravel errors by
# design. This is intentional, documented divergence, not a bug and not
# a missing prometheus_process.rs flag. Prometheus has no per-subquery-
# node step-count cap, and neither of its query resource limits fires
# here: --query.max-samples is a yielded-sample memory budget (these
# sparse 30s-sampled series yield far too few samples to trip it), and
# query_range's outer 11_000-step check never runs because both entries
# are instant queries. Ravel additionally classifies its cap rejection
# as a 422 "execution" budget error (TooManyPoints -> Unsupported), not
# the 400 "bad_data" class Prometheus uses for its own outer resolution
# limit, so the two engines diverge on error class as well as on whether
# they error at all. See
# docs/adrs/0030-promql-subquery-point-cap-divergence.md for the full
# investigation and decision.
#
# These stay `mode: error`: it encodes Ravel's own expected behavior,
# and the per-node cap itself stays covered by ravel-promql's eval.rs
# unit tests. The comparator (comparator.rs) has no mode for "Ravel
# errors by design, Prometheus succeeds"; ADR-0030 records that gap and
# the harness follow-up needed to turn these green rather than
# known-divergent, which is outside this corpus-only change.

name: error_subquery_grid_exceeds_the_point_cap
query: max_over_time(diff_gauge_walk[100d:1s])
kind: instant
time: +0s
mode: error

name: error_nested_subquery_inner_grid_exceeds_the_point_cap
query: avg_over_time(max_over_time(diff_gauge_walk[50d:1s])[1m:1s])
kind: instant
time: +0s
mode: error
